/**
 * Update user balance atomically using SQL transaction or direct UPDATE query.
 */
async function updateBalanceAtomic(chatId, amount, type = 'add', connection = null) {
    const conn = connection || await pool.getConnection();
    try {
        const sign = type === 'add' ? '+' : '-';
        const [result] = await conn.query(
            `UPDATE users SET saldo = saldo ${sign} ? WHERE chatId = ?`,
            [amount, chatId]
        );
        return result.affectedRows > 0;
    } catch (error) {
        console.error(`[DB ERROR] updateBalanceAtomic (${type}):`, error.message);
        throw error;
    } finally {
        if (!connection) conn.release();
    }
}

/**
 * Handle refund process with SQL Transaction and Idempotency check.
 */
async function processRefund(refId, userId, amount, productName, method, reason) {
    const connection = await pool.getConnection();
    try {
        await connection.beginTransaction();

        // 1. Lock the transaction row and check status (Idempotency)
        const [rows] = await connection.query(
            'SELECT userId, price, productName, method, status, is_refunded FROM transactions WHERE trxId = ? FOR UPDATE',
            [refId]
        );

        if (rows.length === 0) {
            logAction(`[REFUND ERROR] Transaksi ${refId} tidak ditemukan.`);
            await connection.rollback();
            return false;
        }

        const trx = rows[0];
        if (trx.is_refunded) {
            logAction(`[REFUND SKIPPED] Transaksi ${refId} sudah pernah di-refund.`);
            await connection.rollback();
            return false;
        }

        // 2. Refund Saldo
        await updateBalanceAtomic(userId, amount, 'add', connection);

        // 3. Update Status Transaksi
        await connection.query(
            'UPDATE transactions SET status = ?, is_refunded = TRUE WHERE trxId = ?',
            ['Gagal (Refunded)', refId]
        );

        await connection.commit();
        logAction(`[REFUND SUCCESS] Ref: ${refId}, User: ${userId}, Amount: Rp${formatRupiah(amount)}`);

        // 4. Notifications
        const userData = await getUserById(userId);
        const newSaldo = userData ? userData.saldo : 'N/A';
        const date = new Date().toISOString().split('T')[0];

        const refundMessage = `
${separator}
✅ *Status Refund Saldo Sukses*
${separator}
\`Code   :\` \`${refId}\`
\`Produk :\` \`${productName || trx.productName || 'N/A'}\`
\`Status :\` \`GAGAL\`
\`Pesan  :\` \`${reason || 'Transaksi Gagal'}\`
\`Refund :\` \`Rp${formatRupiah(amount)}\`
\`Saldo  :\` \`Rp${formatRupiah(newSaldo)}\`
${separator}`;

        await bot.sendMessage(userId, refundMessage, { parse_mode: 'Markdown' }).catch(err => {});

        const groupRefundMessage = `
${separator}
Status Refund Saldo Sukses
${separator}
\`✧ User  :\` \`${escapeHtml(userData?.name || userData?.username || userId)}\`
\`✧ Code  :\` \`${refId}\`
\`✧ Info  :\` \`${productName || trx.productName || 'N/A'}\`
\`✧ Via   :\` \`${method || trx.method || 'N/A'}\`
\`✧ Plus  :\` \`Rp${formatRupiah(amount)}\`
\`✧ Date  :\` \`${date}\`
${separator}`;

        await bot.sendMessage(GROUP_NOTIFICATION_ID, groupRefundMessage, { parse_mode: 'Markdown' }).catch(err => {});

        return true;
    } catch (error) {
        await connection.rollback();
        console.error(`[REFUND ERROR] Ref: ${refId}:`, error.message);
        return false;
    } finally {
        connection.release();
    }
}

function formatWaktu(dateInput) {
    if (!dateInput) return 'N/A';
    const d = new Date(dateInput);

    // Jika input tidak valid
    if (isNaN(d.getTime())) return dateInput;

    const year = d.getFullYear();
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    const hours = String(d.getHours()).padStart(2, '0');
    const minutes = String(d.getMinutes()).padStart(2, '0');

    return `${year}-${month}-${day} ${hours}:${minutes}`;
}

function formatTglJadwal(dateInput, timeInput) {
    if (!dateInput) return 'N/A';

    // 1. Proses Tanggal
    const d = new Date(dateInput);
    if (isNaN(d.getTime())) return dateInput;

    const year = d.getFullYear();
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');

    // 2. Proses Jam (Safety Check untuk String atau Objek)
    let time = "00:00";
    if (timeInput) {
        // Pastikan dikonversi ke string dulu sebelum di-split
        const timeStr = timeInput.toString();
        if (timeStr.includes(':')) {
            time = timeStr.split(':').slice(0, 2).join(':');
        }
    }

    return `${year}-${month}-${day} ${time}`;
}

class CustomQrisPaymentChecker {
    constructor(config) {
        this.auth_token = config.auth_token;
        this.auth_username = config.auth_username;
        this.API_ENDPOINT = 'https://pg-tokovpn.pages.dev/api/orderkuota';
    }

    async checkPaymentStatus(reference, uniqueAmount, startTime) {
        const payload = qs.stringify({
            username: this.auth_username,
            token: this.auth_token
        });

        const expectedKreditString = uniqueAmount.toString();
        const truncatedStartDate = new Date(startTime);
        truncatedStartDate.setSeconds(0, 0);
        const truncatedStartTime = truncatedStartDate.getTime();

        try {
            const response = await axios.post(this.API_ENDPOINT, payload, {
                headers: {
                    "Content-Type": "application/x-www-form-urlencoded",
                    "Origin": "https://payment-tokovpn.pages.dev",
                    "Referer": "https://payment-tokovpn.pages.dev/",
                    "User-Agent": "Mozilla/5.0"
                },
                timeout: 10000
            });

            if (response.status !== 200 || !response.data || !response.data.results) {
                console.log(`[QRIS API ERROR] Response tidak valid:`, response.data);
                return { 
                    success: false, 
                    data: { 
                        status: 'ERROR', 
                        message: `API Response Invalid` 
                    } 
                };
            }

            const foundTransaction = response.data.results.find(trx => {
                const isIN = trx.status === 'IN';

                const actualKreditClean = (trx.kredit || '').toString().replace(/[^0-9]/g, '');
                const isAmountMatch = actualKreditClean === expectedKreditString;

                if (!trx.tanggal) return false;

                const [datePart, timePart] = trx.tanggal.split(' ');
                const [day, month, year] = datePart.split('/');

                const trxTime = new Date(
                    year,
                    month - 1,
                    day,
                    ...(timePart ? timePart.split(':') : [0,0,0])
                ).getTime();

                const isTimeValid = trxTime >= truncatedStartTime;

                return isIN && isAmountMatch && isTimeValid;
            });

            if (foundTransaction) {
                return { 
                    success: true, 
                    data: { 
                        status: 'PAID', 
                        message: `Pembayaran ditemukan: ${foundTransaction.keterangan}` 
                    } 
                };
            } else {
                return { 
                    success: true, 
                    data: { 
                        status: 'UNPAID', 
                        message: 'Nominal belum ditemukan atau belum tercatat.' 
                    } 
                };
            }

        } catch (error) {
            const errorMessage = error.response 
                ? JSON.stringify(error.response.data) 
                : error.message;

            console.error(`[CUSTOM QRIS CHECKER ERROR]`, errorMessage);

            return { 
                success: false, 
                data: { 
                    status: 'ERROR', 
                    message: `API Call Failed: ${errorMessage.substring(0, 100)}` 
                } 
            };
        }
    }
}

function formatRupiah(price) {
    const numPrice = typeof price === 'string' ? parseFloat(price) : price;
    if (isNaN(numPrice) || numPrice === null) return 'N/A';
    return Math.floor(numPrice).toLocaleString('id-ID', { minimumFractionDigits: 0 });
}

function getRuntime() {
    const uptime = Date.now() - startTime;

    const units = [
        { label: "h", value: 1000 * 60 * 60 * 24 },
        { label: "j", value: 1000 * 60 * 60 },
        { label: "m", value: 1000 * 60 },
        { label: "d", value: 1000 }
    ];

    let result = [];
    let remaining = uptime;

    for (const u of units) {
        const val = Math.floor(remaining / u.value);
        if (val > 0 || u.label === "d") result.push(`${val}${u.label}`);
        remaining %= u.value;
    }

    return result.join(" ");
}

function normalizePhoneNumber(phone) {
    if (!phone) return null;
    let normalized = phone.replace(/[^0-9+]/g, '');

    if (normalized.startsWith('+62')) {
        normalized = '0' + normalized.substring(3);
    } else if (normalized.startsWith('62')) {
        normalized = '0' + normalized.substring(2);
    } else if (!normalized.startsWith('08')) {
        return null;
    }

    if (normalized.startsWith('08') && normalized.length >= 10 && normalized.length <= 13) {
        return normalized;
    }

    return null;
}

function calculateSellingPrice(modalPrice) {
    const config = getPriceConfig();
    const MARGIN_PERCENTAGE = config.margin / 100;
    const MAX_FEE = config.max_fee;

    const calculatedFee = modalPrice * MARGIN_PERCENTAGE;
    const finalFee = Math.min(calculatedFee, MAX_FEE);

    let sellingPrice = modalPrice + finalFee;
    return Math.round(sellingPrice);
}

function scanProductsInText(text) {
    const foundProducts = [];
    const seenNames = new Set();
    let allAvailableProducts = [];

    for (const category in groupedProducts) {
        for (const brand in groupedProducts[category]) {
            for (const type in groupedProducts[category][brand]) {
                allAvailableProducts.push(...groupedProducts[category][brand][type].map(p => ({ type: 'PPOB', data: p })));
            }
        }
    }

    allAvailableProducts.forEach(item => {
        const index = text.indexOf(item.data.name);
        if (index !== -1 && !seenNames.has(item.data.name)) {
            foundProducts.push({
                ...item,
                pos: index
            });
            seenNames.add(item.data.name);
        }
    });

    return foundProducts.sort((a, b) => a.pos - b.pos);
}

    }

    const totalItems = type === 'trx' ? await countUserTransactions(targetId) : await countUserTopups(targetId);
    const totalPages = Math.ceil(totalItems / MAX_ITEMS_PER_PAGE);
    const startIdx = (page - 1) * MAX_ITEMS_PER_PAGE;

    const dataArray = type === 'trx'
        ? await getUserTransactions(targetId, MAX_ITEMS_PER_PAGE, startIdx)
        : await getUserTopups(targetId, MAX_ITEMS_PER_PAGE, startIdx);

    const title = type === 'trx' ? 'Riwayat Pesanan' : 'Riwayat TopUp';
    const detailAction = type === 'trx' ? 'trx_admin_detail' : 'topup_admin_detail';
    const actionPrefix = type === 'trx' ? 'admin_view_trx' : 'admin_view_topup';
    const sep = escapeHtml(separator);

    let nextText = '';
    nextText += `${sep}\n`;
    nextText += `<b>👥 ${escapeHtml(targetData.name)}</b>\n`;
    nextText += `<b>🔎 ${escapeHtml(title)}</b> [ ${totalItems} Item ]\n`;
    nextText += `Halaman <b>${page}</b> dari <b>${totalPages || 1}</b>\n`;
    nextText += `${sep}\n`;

    const detailButtons = [];
    const itemLines = [];

    if (dataArray.length > 0) {
        dataArray.forEach((item, index) => {
            const itemNumber = startIdx + index + 1;
            const statusIcon = item.status === 'Sukses' ? '✅' : (item.status && item.status.startsWith && item.status.startsWith('Gagal') ? '⛔️' : '⏳');

            const mainInfo = type === 'trx'
                ? escapeHtml(item.productName || '-')
                : `Rp${escapeHtml(formatRupiah(item.amount || item.price || 0).toString())} - ${escapeHtml(item.method || '-')}`;

            itemLines.push(`<b>${itemNumber}.</b> ${statusIcon} ${mainInfo}`);
            detailButtons.push({ text: `${itemNumber}`, callback_data: `${detailAction}_${targetId}_${item.trxId || item.id}_${page}` });
        });

        nextText += itemLines.join('\n') + `\n${sep}\nPilih Angka Untuk Detail Informasi!\n${sep}\n`;
    } else {
        nextText += `<i>Tidak ada riwayat yang tercatat.</i>\n`;
    }

    const detailRows = [];
    for (let i = 0; i < detailButtons.length; i += 3) {
        detailRows.push(detailButtons.slice(i, i + 3));
    }

    const paginationButtons = [];
    if (page > 1) {
        paginationButtons.push({ text: `⬅️ Hal. ${page - 1}`, callback_data: `${actionPrefix}_${targetId}_${page - 1}` });
    }
    if (page < totalPages) {
        paginationButtons.push({ text: `Hal. ${page + 1} ➡️`, callback_data: `${actionPrefix}_${targetId}_${page + 1}` });
    }

    const keyboard = [
        ...detailRows,
        ...(paginationButtons.length ? [paginationButtons] : []),
        [{ text: '↩️ Kembali Ke Detail User', callback_data: `admin_user_detail_${targetId}_1` }]
    ];

    const options = {
        chat_id: chatId,
        message_id: messageId,
        parse_mode: 'HTML',
        reply_markup: { inline_keyboard: keyboard },
        disable_web_page_preview: true
    };

    try {
        if (messageId) {
            await bot.editMessageText(nextText, options);
        } else {
            await bot.sendMessage(chatId, nextText, options);
        }
    } catch (e) {
        console.error(`[ADMIN TRX HISTORY ERROR] Gagal mengirim riwayat Admin: ${e.message}`);
        await bot.sendMessage(chatId, `⛔️ Gagal memuat riwayat: ${escapeHtml(e.message)}`, { parse_mode: 'HTML' });
    }
}

function generateDigiflazzSign(refId) {
    const rawString = DIGIFLAZZ_USERNAME + DIGIFLAZZ_API_KEY + refId;
    return crypto.createHash('md5').update(rawString).digest('hex');
}

function makePricelistSign() {
    return crypto
        .createHash("md5")
        .update(DIGIFLAZZ_USERNAME + DIGIFLAZZ_API_KEY + "pricelist")
        .digest("hex");
}

async function updatePriceList(isInitialLoad = false) {
    const cmd = "prepaid";
    const sign = makePricelistSign();
    const payload = { cmd, username: DIGIFLAZZ_USERNAME, sign };

    try {
        const response = await axios.post(PRICELIST_ENDPOINT, payload, {
            headers: { "Content-Type": "application/json" },
        });

        if (response.data?.data?.rc === "83") {
            logAction("⚠️ Digiflazz: Sudah mencapai limitasi pengecekan pricelist.");
            return { success: false, message: "Sudah mencapai limitasi pengecekan pricelist (RC 83)." };
        }

        const products = Array.isArray(response.data?.data) ? response.data.data : (Array.isArray(response.data?.data?.data) ? response.data.data.data : []);

        if (!products.length) {
            logAction("⚠️ Digiflazz: Tidak ada data produk, respons tidak sesuai harapan.");
            return { success: false, message: "Tidak ada data produk dari API Digiflazz." };
        }

        const filePath = path.join(__dirname, PRICELIST_PREPAID_FILE);
        fs.writeFileSync(filePath, JSON.stringify(products, null, 2), "utf-8");
        loadAndGroupProducts();

        const successMessage = `✅ Daftar harga ${cmd.toUpperCase()} berhasil diperbarui (${products.length} produk).`;
        logAction(successMessage);
        return { success: true, message: successMessage };

    } catch (error) {
        const errorMessage = `⛔️ Gagal ambil daftar harga: ${error.response?.data?.data?.message || error.message}`;
        console.error(errorMessage);
        return { success: false, message: errorMessage };
    }
}

async function callDigiflazzAPI(refId, sku, customerNo) {
    const sign = generateDigiflazzSign(refId);
    const requestData = {
        username: DIGIFLAZZ_USERNAME,
        buyer_sku_code: sku,
        customer_no: customerNo,
        ref_id: refId,
        sign: sign
    };

    logAction(`API: Mempersiapkan request untuk Ref: ${refId}`);

    try {
        const response = await axios.post(DIGIFLAZZ_ENDPOINT, requestData, {
            headers: { 'Content-Type': 'application/json' }
        });

        logAction(`API: Respon sukses (Status ${response.status}) diterima untuk Ref: ${refId}`);

        if (response.data && response.data.data) {
            return response.data.data;
        } else {
            throw new Error(`Invalid API response structure. Response: ${JSON.stringify(response.data)}`);
        }
    } catch (error) {
        const errorMessage = error.response ? `HTTP Error ${error.response.status}: ${JSON.stringify(error.response.data)}` : error.message;
        console.error(`[DEBUG: API ERROR] Digiflazz call failed for ${refId}:`, errorMessage);
        logAction(`API: Gagal memanggil API untuk Ref: ${refId}. Error: ${errorMessage.substring(0, 50)}...`);

        return {
            ref_id: refId,
            customer_no: customerNo,
            buyer_sku_code: sku,
            message: `Koneksi Error, Hubungi atau teruskan pesan ini ke admin segera!`,
            // message: `Koneksi Error: ${errorMessage}`,
            status: "Gagal",
            rc: "99"
        };
    }
}

async function startQrisCheck(chatId, userId, reference, uniqueAmount, type, qrisMessageId) {
    const timeout = 10 * 30 * 1000;
    const startTime = Date.now();
    let finalStatus = 'Gagal (Expired)';
    let isPaid = false;

    try {
        const userData = await getUserById(userId);
        if (!userData) throw new Error("User data not found");

        await updateUser(userId, { isMonitoringQris: true });

        while (Date.now() - startTime < timeout) {
            const updatedUserData = await getUserById(userId);
            if (!updatedUserData || !updatedUserData.isMonitoringQris) {
                finalStatus = 'Dibatalkan Pengguna';
                logAction(`QRIS CHECK: Monitoring Ref ${reference} dihentikan oleh pengguna.`);
                break;
            }

            console.log(`[QRIS CHECK LOG] Checking Ref ${reference} with Unique Amount ${uniqueAmount}...`);
            const result = await paymentChecker.checkPaymentStatus(reference, uniqueAmount, startTime);

            if (result.success && result.data.status === 'PAID') {
                finalStatus = 'Sukses';
                isPaid = true;
                break;
            }
            await new Promise(resolve => setTimeout(resolve, 15000));
        }
    } catch (error) {
        console.error(`[QRIS CHECK ERROR] Ref ${reference}:`, error.message);
        finalStatus = 'Gagal (Error API)';
    } finally {
        await updateUser(userId, { isMonitoringQris: false });
        logAction(`QRIS CHECK: Monitoring Ref ${reference} selesai. Status: ${finalStatus}`);
    }

    const userData = await getUserById(userId);
    if (!userData) return;

    let transaction;

    if (type === 'topup') {
        transaction = await getTopupByTrxId(reference);
    } else {
        transaction = await getTransactionByTrxId(reference);
    }

    if (!transaction || transaction.status !== 'Pending') return;

    const originalAmount = transaction.amount || transaction.price;
    let apiStatus = finalStatus;
    let apiMessage = `Pembayaran ${finalStatus}`;

    try {
        await bot.deleteMessage(chatId, qrisMessageId);
    } catch (e) { }

    if (type === 'topup') {
        if (isPaid) {
            apiStatus = 'Sukses';
            await updateTopup(reference, { status: apiStatus });

            await updateBalanceAtomic(userId, originalAmount, 'add');

            const updatedTopup = await getTopupByTrxId(reference);
            await sendTrxNotificationToGroup(updatedTopup, 'topup', userData);
        } else {
            await updateTopup(reference, { status: finalStatus });
        }
    } else if (type === 'trx' && isPaid) {
        const apiResult = await callDigiflazzAPI(reference, transaction.productCode, transaction.phoneNumber);
        apiStatus = apiResult.status;
        apiMessage = apiResult.message;

        await updateTransaction(reference, {
            status: apiStatus,
            sn: apiResult.sn || apiResult.message
        });

        if (apiStatus === 'Gagal') {
            await processRefund(reference, userId, originalAmount, transaction.productName, transaction.method, apiMessage);
        }

        if (apiStatus === 'Sukses') {
            const updatedTrx = await getTransactionByTrxId(reference);
            await sendTrxNotificationToGroup(updatedTrx, 'trx', userData);
        }
    } else {
        if (type === 'trx') {
            await updateTransaction(reference, { status: finalStatus });
        } else {
            await updateTopup(reference, { status: finalStatus });
        }
    }

    const updatedUserData = await getUserById(userId);
    const finalTransaction = type === 'trx' ? await getTransactionByTrxId(reference) : await getTopupByTrxId(reference);
    const finalTrxStatus = finalTransaction.status || finalStatus;

    let notificationIcon = '⛔️';
    if (finalTrxStatus.includes('Sukses')) {
        notificationIcon = '✅';
    } else if (finalTrxStatus.includes('Pending')) {
        notificationIcon = '⏳';
    }

    let apiStatusDetail = '';
    if (type === 'trx' && isPaid && apiStatus !== 'Sukses' && apiStatus !== finalStatus) {
        apiStatusDetail = `\n\`Pesan  :\` \`Transaksi ${apiStatus}\``;
    }

    const notificationMessage = `
${separator}
${notificationIcon} Status Pembayaran User
${separator}
\`ID TRX :\` \`${reference}\`
\`Status :\` \`${finalTrxStatus.toUpperCase()}\`${apiStatusDetail}
${type === 'topup' ? `\`TopUp  :\` \`Rp${formatRupiah(originalAmount)}\`` : `\`Harga  :\` \`Rp${formatRupiah(originalAmount)}\``} ${type === 'topup' && finalTrxStatus.includes('Sukses') ? `\`\nSaldo  :\` \`Rp${formatRupiah(updatedUserData.saldo)}\`` : ''}
${separator}`;

    await bot.sendMessage(chatId, notificationMessage, { parse_mode: 'Markdown' });

    if (!(type === 'trx' && (apiStatus === 'Sukses' || apiStatus === 'Pending'))) {
        await delay(2000);
        await sendMainMenuFinal(chatId, updatedUserData);
    }
}

async function findPendingQris(userId) {
    try {
        const userData = await getUserById(userId);
        if (!userData || !userData.isMonitoringQris) return null;

        const [trxRows] = await pool.query(
            "SELECT * FROM transactions WHERE userId = ? AND method = 'QRIS' AND status = 'Pending' LIMIT 1",
            [userId]
        );

        if (trxRows.length > 0) return trxRows[0];

        const [topupRows] = await pool.query(
            "SELECT * FROM topups WHERE userId = ? AND method = 'QRIS' AND status = 'Pending' LIMIT 1",
            [userId]
        );

        if (topupRows.length > 0) return topupRows[0];

        return null;
    } catch (error) {
        console.error('[ERROR] findPendingQris:', error.message);
        return null;
    }
}

function formatResult(text) {
    return text
        .replace(/(Paket\s+\d+:\s+)Paket\s+/gm, "$1")
        .replace(/^Nomor:/gm, "📱 *Nomor:*")
        .replace(/^Tipe Kartu:/gm, "💳 *Tipe Kartu:*")
        .replace(/^Status 4G:/gm, "📶 *Status 4G:*")
        .replace(/^Status Dukcapil:/gm, "🧾 *Status Dukcapil:*")
        .replace(/^Umur Kartu:/gm, "⏳ *Umur Kartu:*")
        .replace(/^Masa Aktif:/gm, "📆 *Masa Aktif:*")
        .replace(/^Masa Berakhir Tenggang:/gm, "⛔️ *Akhir Tenggang:*")
        .replace(/^Detail Kuota:/gm, "💎 *Detail Kuota:*")
        .replace(/^Paket\s+(\d+):/gm, "🗂️ *Paket $1:*")
        .replace(/^\s*Benefit\s*:/gm, "🎁 *Benefit:*")
        .replace(/^Aktif Hingga:/gm, "⏳ *Aktif Hingga:*")
        .replace(/^- /gm, "🔹 ");
}

async function cekXL(nomor) {
    try {
        const url = `https://bendith.my.id/end.php?check=package&number=${nomor}&version=2`;

        const res = await axios.get(url, {
            headers: {
                "accept": "application/json",
                "user-agent": "Mozilla/5.0"
            }
        });

        if (!res.data || !res.data.success) {
            return {
                success: false,
                data: "Status Pengecekan Gagal."
            };
        }

        const subs = res.data.data?.subs_info;
        const packages = res.data.data?.package_info?.packages || [];

        if (!subs) {
            return {
                success: false,
                data: "Data pelanggan tidak ditemukan."
            };
        }

        let rawText = 
`Nomor: ${subs.msisdn}
Tipe Kartu: ${subs.operator}
Status 4G: ${subs.net_type}
Status Dukcapil: ${subs.id_verified}
Umur Kartu: ${subs.tenure}
Masa Aktif: ${subs.exp_date}
Masa Berakhir Tenggang: ${subs.grace_until}

Detail Kuota:
`;

        if (packages.length > 0) {
            packages.forEach((pkg, index) => {
                rawText += `
Paket ${index + 1}: ${pkg.name}
Aktif Hingga: ${pkg.expiry}
Benefit:
`;

                if (pkg.quotas && pkg.quotas.length > 0) {
                    pkg.quotas.forEach(q => {
                        rawText += `- ${q.name}\n`;
                        rawText += `  Sisa: ${q.remaining} / ${q.total} (${q.percent}%)\n`;
                    });
                } else {
                    rawText += `- Tidak ada detail kuota\n`;
                }
            });
        } else {
            rawText += `Tidak ada paket aktif.\n`;
        }

        return {
            success: true,

const createKeyboard = (items, backData, twoColumns = true) => {
    const keyboard = [];
    let row = [];
    const maxCols = twoColumns ? 2 : 1;

    for (let i = 0; i < items.length; i++) {
        row.push(items[i]);
        if (row.length === maxCols || i === items.length - 1) {
            keyboard.push(row);
            row = [];
        }
    }

    if (backData) {
        keyboard.push([{ text: '↩️ Kembali', callback_data: backData }]);
    }
    if (backData) {
        keyboard.push([{ text: '🏠 Menu Utama', callback_data: 'main_menu' }]);
    }

    return keyboard;
};

async function showMonthPicker(chatId, messageId) {

        buttons.push({
            text: monthShort,
            callback_data: `sch_m_${d.getMonth()}_${d.getFullYear()}`
        });
    }

async function showDayPicker(chatId, messageId, month, year) {

async function showTimePicker(chatId, messageId, d, m, y) {

async function renderJadwalList(chatId, message_id, page = 1) {
    const totalSchedules = await countUserSchedules(chatId);

    if (totalSchedules === 0) {
        return await bot.editMessageText(
            `${separator}\n📅 *Daftar Jadwal Saya*\n${separator}\n` +
            `⚠️ Belum Ada Jadwal\n` +
            `_Belum ada riwayat penjadwalan._\n${separator}`,
            {
                chat_id: chatId,
                message_id: message_id,
                parse_mode: 'Markdown',
                reply_markup: {
                    inline_keyboard: [
                        [{ text: '🏠 Menu Utama', callback_data: 'main_menu' }]
                    ]
                }
            }
        );
    }

    const totalPages = Math.ceil(totalSchedules / MAX_ITEMS_PER_PAGE);
    const startIdx = (page - 1) * MAX_ITEMS_PER_PAGE;
    const pageData = await getUserSchedules(chatId, MAX_ITEMS_PER_PAGE, startIdx);

    let nextText = `${separator}\n<blockquote><code>Menu •¯•» Jadwal Aktif</code></blockquote>\n${separator}\n<code>Daftar Jadwal [ ${page} / ${totalPages} ]</code>\n${separator}\n`;

    pageData.forEach((j, index) => {
        const productResult = findProductByName(j.productName);
        const latestProduct = productResult ? productResult.data : null;
        const displayPrice = latestProduct ? latestProduct.sellingPrice : j.price;

        nextText += `<b>${startIdx + index + 1}. ✅ ${escapeHtml(j.productName)}</b>\n` +
            `<code>Tujuan :</code> <code>${j.phoneNumber}</code>\n` +
            `<code>Harga  :</code> <code>Rp${formatRupiah(displayPrice)}</code>\n` +
            `<code>Waktu  :</code> <code>${formatTglJadwal(j.executeDate, j.executeTime)}</code>\n${separator}\n`;
    });

    const numButtons = pageData.map((j, index) => ({
        text: `${startIdx + index + 1}`,
        callback_data: `sched_del_confirm_${j.id}_${page}`
    }));

    const numRows = [];
    for (let i = 0; i < numButtons.length; i += 3) numRows.push(numButtons.slice(i, i + 3));

    const navButtons = [];
    if (page > 1) navButtons.push({ text: '⬅️', callback_data: `sched_list_${page - 1}` });
    if (page < totalPages) navButtons.push({ text: '➡️', callback_data: `sched_list_${page + 1}` });

    const keyboard = [...numRows, ...(navButtons.length ? [navButtons] : []), [{ text: '🏠 Menu Utama', callback_data: 'main_menu' }]];

    return await bot.editMessageText(nextText, {
        chat_id: chatId,
        message_id: message_id,
        parse_mode: 'HTML',
        reply_markup: { inline_keyboard: keyboard }
    });
}

bot.on('callback_query', async (callbackQuery) => {
    const msg = callbackQuery.message;
    let data = callbackQuery.data;
    const chatId = msg.chat.id;
    const messageId = msg.message_id;
    const user = callbackQuery.from;
    const userData = await initializeUser(user);
    const userId = user.id;

    console.log(`\n[DEBUG] Klik Terdeteksi: ${data} | User: ${chatId}`);

    if (isMaintenanceMode || isMaintenanceModeOtomatis) {
        const maintMsg = isMaintenanceMode ? globalMaintenanceMessage : "Bot sedang dalam mode maintenance harian (23:30 - 00:30). Mohon coba lagi nanti.";
        await bot.answerCallbackQuery(callbackQuery.id, {
            text: `🛠️ ${maintMsg}`,
            show_alert: true
        });
        return;
    }

    const currentTime = Date.now();
    const lastClickTime = callbackCooldowns[userId] || 0;

    if (currentTime - lastClickTime < CALLBACK_COOLDOWN_MS) {
        await bot.answerCallbackQuery(callbackQuery.id, {
            text: "⚠️ Mohon tunggu sejenak sebelum menekan lagi.",
            show_alert: false
        });
        return;
    }
    callbackCooldowns[userId] = currentTime;

    if (!data.startsWith('buy_') && !data.startsWith('confirm_tx_') && !data.startsWith('admin_') &&
        !data.startsWith('info_qris_') &&
        data !== 'admin_cancel_broadcast' && data !== 'cancel_tx' &&
        !data.startsWith('sched_') && !data.startsWith('sch_')) {
        await bot.answerCallbackQuery(callbackQuery.id).catch(() => { });
    }

    if (data.startsWith('favbuy_')) {
        const b64Part = data.split('_')[1];
        let targetProductName = null;
        const favorites = await getFavoriteProducts(userId);
        const foundFav = favorites.find(f => Buffer.from(f.name).toString('base64').substring(0, 30) === b64Part);

        if (foundFav) {
            targetProductName = foundFav.name;
        } else {
            let foundInGlobal = null;
            for (const cat in groupedProducts) {
                for (const brd in groupedProducts[cat]) {
                    for (const typ in groupedProducts[cat][brd]) {
                        const match = groupedProducts[cat][brd][typ].find(p =>
                            Buffer.from(p.name).toString('base64').substring(0, 30) === b64Part
                        );
                        if (match) { foundInGlobal = match.name; break; }
                    }
                    if (foundInGlobal) break;
                }
                if (foundInGlobal) break;
            }
            targetProductName = foundInGlobal;
        }

        if (!targetProductName) {
            return bot.answerCallbackQuery(callbackQuery.id, { text: "⛔️ Produk sudah tidak tersedia atau ID expired.", show_alert: true });
        }

        const productInfo = findProductByName(targetProductName);
        if (!productInfo) {
            return bot.answerCallbackQuery(callbackQuery.id, { text: "⛔️ Data produk tidak ditemukan di pusat.", show_alert: true });
        }

        const p = productInfo.data;
        const currentPrice = p.sellingPrice;

        userStates[chatId] = {
            step: 'AWAITING_PHONE',
            productCode: p.code,
            productName: p.name,
            price: currentPrice
        };

        let panduanInput = `_Kirim nomor tujuan [ contoh: 0812345xxxx ].\nJika No hp: ✅ 08xx ✅ 628xx ✅ +628xx_`;
        const nameLower = p.name.toLowerCase();

        if (nameLower.includes("free fire")) {
            panduanInput = `_Kirimkan ID Akun Free Fire Anda.\nContoh: 123456789_`;
        } else if (nameLower.includes("mobile legend") || nameLower.includes("mobilelegend")) {
            panduanInput = `_Kirimkan ID + Server Mobile Legends Anda.\nContoh: 12345678(1234) tanpa tanda kurung_`;
        }

        const instructionMessage = `
${separator}
*Detail Pembelian Produk*
${separator}
\`Produk :\` \`${escapeHtml(p.name)}\`
\`Code   :\` \`${escapeHtml(p.code)}\`
\`Detail :\` \`${escapeHtml(p.desc)}\`
\`Harga  :\` \`Rp${formatRupiah(currentPrice)}\`
${separator}
📝 *Transaksi diproses otomatis. Pastikan nomor/ID tujuan benar. Kesalahan input di luar tanggung jawab penjual.*

${panduanInput}
_Setelah dikirim, konfirmasi Bayar atau Batalkan Proses._
${separator}`;

        try {

                message_id: messageId,
                parse_mode: 'Markdown',
                reply_markup: { inline_keyboard: [[{ text: '⛔️ Batalkan Proses', callback_data: 'cancel_tx' }]] }

    else if (data.startsWith('buy_')) {

            userStates[chatId] = {

${separator}
*Detail Pembelian Produk*
${separator}
\`Produk :\` \`${escapeHtml(product.name)}\`
\`Code   :\` \`${escapeHtml(product.code)}\`
\`Detail :\` \`${escapeHtml(product.desc || '-')}\`
\`Harga  :\` \`Rp${formatRupiah(product.sellingPrice)}\`
${separator}
📝 *Transaksi diproses otomatis. Pastikan nomor/ID tujuan benar. Kesalahan input di luar tanggung jawab penjual.*

${panduanInput}
_Setelah dikirim, konfirmasi Bayar atau Batalkan Proses._
${separator}`;

                const apiResult = await callDigiflazzAPI(refId, state.productCode, state.phoneNumber);

                    await connection.rollback(); // Rollback balance subtraction

                    snDetail = 'N/A';

                        userId: chatId,
                        trxId: refId,

                        apiMessage: apiMessage,

\`Saldo  :\` \`Rp${formatRupiah(currentSaldo)}\`
${separator}
${status === 'Sukses' ? 'Pemesanan diproses.' : status === 'Pending' ? 'Pesanan Sedang Diproses Otomatis.' : 'Transaksi Dibatalkan atau Gagal.'}
${separator}`;

                    try { await bot.deleteMessage(chatId, messageId); } catch (e) { }
                    const updatedUser = await getUserById(chatId);
                    await sendMainMenuFinal(chatId, updatedUser);
                }

            } catch (error) {
                console.error("[ERROR TRX BOT]:", error.message);
                await bot.sendMessage(chatId, "⛔️ Terjadi kesalahan sistem saat memproses transaksi.");
            } finally {
                // WAJIB: Buka kunci agar user bisa melakukan transaksi lagi nanti
                delete userProcessingLock[userId];
            }
            return;
        }
    }

    else if (data === 'confirm_tx_qris') {
        const state = userStates[chatId];
        const userId = callbackQuery.from.id;

        // 1. Anti-Spam Lock: Cek apakah user sedang dalam proses
        if (userProcessingLock[userId]) {
            return bot.answerCallbackQuery(callbackQuery.id, {
                text: "⚠️ Sedang menyiapkan QRIS, mohon tunggu...",
                show_alert: true
            });
        }

        // 2. Proteksi State
        if (!state || !state.productCode || !state.price) {
            return bot.editMessageText(
                `⚠️ *Sesi Berakhir Otomatis*\n\nSesi pembelian terputus karena bot baru saja dimulai ulang. Silakan pilih produk kembali dari Menu Utama.`,
                {
                    chat_id: chatId,
                    message_id: messageId,
                    parse_mode: 'Markdown',
                    reply_markup: {
                        inline_keyboard: [[{ text: '🏠 Menu Utama', callback_data: 'main_menu' }]]
                    }
                }
            ).catch(() => { });
        }

        if (state && state.step === 'AWAITING_CONFIRM') {
            // Aktifkan pengunci
            userProcessingLock[userId] = true;

            try {
                // Cek apakah ada QRIS yang masih menggantung
                const pending = await findPendingQris(userData.chatId);
                if (pending) {
                    await sendPendingQrisWarning(chatId, userData, pending);
                    await bot.answerCallbackQuery(callbackQuery.id, { text: "⚠️ Masih Ada Transaksi QRIS Pending!", show_alert: true });
                    // Langsung keluar dari blok try jika ada pending, lock akan dibuka di finally
                    return;

                // Alert proses
                await bot.answerCallbackQuery(callbackQuery.id, { text: "⏳ Mempersiapkan QRIS..." });

                const sellingPrice = parseFloat(state.price);
                const reference = `TRX${Date.now()}`;

                const currentUserId = callbackQuery.from.id;

                // Generate Gambar QRIS
                const qrString = qrisGen.generateQrString(uniqueAmount, reference);
                const qrBuffer = await qrisGen.generateQRWithLogo(qrString);

                const finalMessage = `
${separator}
💳 *Scan QRIS Untuk Bayar*
${separator}
\`Produk :\` \`${escapeHtml(state.productName)}\`

                    userId: userData.chatId,
                    trxId: reference,
                    time: Date.now(),
                    productName: state.productName,
                    productCode: state.productCode,
                    phoneNumber: state.phoneNumber,
                    price: sellingPrice,
                    method: 'QRIS',
                    status: 'Pending',
                    uniqueAmount: uniqueAmount
                });

                // Hapus pesan tombol konfirmasi sebelumnya
                try { await bot.deleteMessage(chatId, messageId); } catch (e) { }

                // Jalankan checker mutasi
                startQrisCheck(chatId, currentUserId, reference, uniqueAmount, 'trx', sentQr.message_id);

                // Reset state user
                delete userStates[chatId];

            } catch (error) {
                console.log(`\n[FATAL ERROR QRIS]: ${error.message}`);
                await bot.sendMessage(chatId, `⛔️ Gagal memproses QRIS: ${error.message}`);
            } finally {
                // WAJIB: Buka kunci agar user bisa klik tombol lain nanti
                delete userProcessingLock[userId];
            }

    }

    if (showAlert) {
        await bot.answerCallbackQuery(callbackQuery.id, {
            text: showAlert, show_alert: true
        });
    }

    if (data === 'main_menu') {
        await sendMainMenuFinal(chatId, userData, messageId);
        return;
    }

    else if (data.startsWith('cat_')) {
        const topups = await getUserTopups(userData.chatId);
        const transactions = await getUserTransactions(userData.chatId);

        const totalTopup = topups.filter(t => t.status === 'Sukses').reduce((sum, t) => sum + parseFloat(t.amount), 0);
        const totalTrx = transactions.filter(t => t.status === 'Sukses').reduce((sum, t) => sum + parseFloat(t.price), 0);

        const category = data.substring(4);
        const brands = groupedProducts[category] ? Object.keys(groupedProducts[category]) : [];
        const newBackData = 'main_menu';

        const nextMenuText = `
█▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀█

█░░║║║╠─║─║─║║║║║╠─░░█

█▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄█

๑۞๑ ${escapeHtml(userData.name)}
${separator}
<code>Chat ID  :</code> <code>${userData.chatId}</code>
<code>Balance  :</code> <code>Rp${formatRupiah(userData.saldo)}</code>
${separator}
<code>TopUp    :</code> <code>Rp${formatRupiah(totalTopup)}</code>
<code>Done TRX :</code> <code>Rp${formatRupiah(totalTrx)}</code>
${separator}
<blockquote><code>Kategori •¯•» ${escapeHtml(category)}</code></blockquote>
${separator}
<code>Pilih Brand Andalan Kalian!</code>
${separator}
    `;
        const brandItems = brands.map(brand => ({
            text: brand,
            callback_data: `brand_${escapeHtml(category)}_${escapeHtml(brand)}`

        const mainKeyboard = [
            ...brandItems.reduce((rows, item, index) => {
                if (index % 2 === 0) rows.push([]);
                rows[rows.length - 1].push(item);

    else if (data.startsWith('brand_')) {
        const topups = await getUserTopups(userData.chatId);
        const transactions = await getUserTransactions(userData.chatId);

        const totalTopup = topups.filter(t => t.status === 'Sukses').reduce((sum, t) => sum + parseFloat(t.amount), 0);
        const totalTrx = transactions.filter(t => t.status === 'Sukses').reduce((sum, t) => sum + parseFloat(t.price), 0);

        const parts = data.split('_');
        const category = parts[1];
        const brand = parts[2];
        const types = groupedProducts[category] && groupedProducts[category][brand] ? Object.keys(groupedProducts[category][brand]) : [];
        const newBackData = `cat_${category}`;

        const nextMenuText = `
█▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀▀█
█░░╦─╦╔╗╦─╔╗╔╗╔╦╗╔╗░░█
█░░║║║╠─║─║─║║║║║╠─░░█
█░░╚╩╝╚╝╚╝╚╝╚╝╩─╩╚╝░░█
█▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄▄█

๑۞๑ ${escapeHtml(userData.name)}
${separator}
<code>Chat ID  :</code> <code>${userData.chatId}</code>
<code>Balance  :</code> <code>Rp${formatRupiah(userData.saldo)}</code>
${separator}
<code>TopUp    :</code> <code>Rp${formatRupiah(totalTopup)}</code>
<code>Done TRX :</code> <code>Rp${formatRupiah(totalTrx)}</code>
${separator}
<blockquote><code>Brand •¯•» ${escapeHtml(brand)}</code></blockquote>
${separator}
<code>Pilih Produk Andalan Kalian!</code>
${separator}
    `;

        const brand = parts[2];
        const type = parts[3];

        const products = groupedProducts[category][brand][type] || [];
        const totalPages = Math.ceil(products.length / MAX_ITEMS_PER_PAGE);
        const startIdx = (page - 1) * MAX_ITEMS_PER_PAGE;

        const pageProducts = products.slice(startIdx, endIdx);

                chat_id: chatId,
                message_id: messageId,
                parse_mode: 'HTML',
                reply_markup: { inline_keyboard: keyboard },
                disable_web_page_preview: true

        } catch (error) {
            console.error("Gagal mengedit pesan (type_):", error.message);
            const sentMessage = await bot.sendMessage(chatId, nextText, {
                parse_mode: 'HTML',
                reply_markup: { inline_keyboard: keyboard },
                disable_web_page_preview: true
            });
            await updateUser(userData.chatId, { lastMessageId: sentMessage.message_id });
        }
        return;
    }
});

bot.on('message', async (msg) => {
    const chatId = msg.chat.id;
    const text = msg.text ? msg.text.toLowerCase().trim() : '';
    const rawText = msg.text ? msg.text.trim() : '';
    const isCommand = text.startsWith('/');
    const state = userStates[chatId];
    const userData = await initializeUser(msg.from);
    if (isCommand) {
        const adminId = ADMIN_ID;
        const parts = rawText.split(/\s+/);
        const command = parts[0].toLowerCase();

        if (chatId == adminId) {
            if (command === '/pesan' && parts.length > 1) {
                const content = rawText.substring(7).trim();
                const [targetIdRaw, ...messageParts] = content.split('|');
                const targetId = targetIdRaw ? targetIdRaw.trim() : null;
                const messageToSend = messageParts.join('|').trim();

                if (!targetId || !messageToSend) {
                    return bot.sendMessage(chatId, "⛔️ Format salah!\nContoh: `/pesan 12345678|Halo kak`", { parse_mode: 'Markdown' });
                }

                const targetUser = await getUserById(targetId);
                if (!targetUser) {
                    return bot.sendMessage(chatId, "⛔️ User ID tidak ditemukan.");
                }

                try {
                    await bot.sendMessage(targetId, `📩 *Pesan dari Admin : *\n\n${messageToSend}`, { parse_mode: 'Markdown' });

                    const successAdmin = `
${separator}
✅ *Pesan Admin Terkirim*
${separator}
\`Nama     :\` \`${escapeHtml(targetUser.name)}\`
\`Chat ID  :\` \`${targetId}\`
\`Status   :\` \`TERKIRIM\`
${separator}`;

                    await bot.sendMessage(chatId, successAdmin, { parse_mode: 'Markdown' });

            }
        }

        const isInputValidAmount = state.step === 'AWAITING_TOPUP_AMOUNT' && sanitizedAmount >= 1;

        if (isCommand || (!isInputValidPhone && !isInputValidAmount)) {
            await updateUser(userData.chatId, { isMonitoringQris: false });
            delete userStates[chatId];

            if (isCommand) { return; }

            try { await bot.deleteMessage(chatId, userData.lastMessageId); } catch (e) { }

            await bot.sendMessage(chatId, "⚠️ Input dibatalkan otomatis karena format salah atau menerima pesan non-transaksi.", { parse_mode: 'Markdown' });
            await sendMainMenuFinal(chatId, userData);
            return;
        }
    }

    if (state) {
        if (state.step === 'AWAITING_TOPUP_AMOUNT') {
            const amount = sanitizeRupiahInput(rawText);

            if (isNaN(amount) || amount < 1000) {
                try { await bot.deleteMessage(chatId, userData.lastMessageId); } catch (e) { }

                await bot.sendMessage(chatId, "⛔️ Nominal Harus Angka dan Minimal *Rp1000*. Mohon Masukkan Nominal yang Benar:", { parse_mode: 'Markdown' });
                return;
            }

            try { await bot.deleteMessage(chatId, userData.lastMessageId); } catch (e) { }

            const loadingMessage = await bot.sendMessage(chatId, "⏳ Mempersiapkan pembayaran QRIS...", { parse_mode: 'Markdown' });
            const { uniqueAmount, uniqueNumber } = generateUniqueQrisAmount(amount);
            const amountToShow = amount;
            const reference = `TOPUP${Date.now()}`;
            const userId = userData.chatId;

            try {
                const qrString = qrisGen.generateQrString(uniqueAmount, reference);
                const qrBuffer = await qrisGen.generateQRWithLogo(qrString);
                const qrImagePath = path.join(__dirname, `qr_topup_${chatId}_${reference}.png`);
                fs.writeFileSync(qrImagePath, qrBuffer);

                try { await bot.deleteMessage(chatId, loadingMessage.message_id); } catch (e) { }

                const sentQr = await bot.sendPhoto(chatId, qrImagePath, {
                    caption: `💳 <b>Scan QRIS Untuk Bayar</b>\n\n` +
                        `<code>ID TRX :</code> <code>${reference}</code>\n` +
                        `<code>TopUp  :</code> <code>Rp${formatRupiah(amountToShow)}</code>\n` +
                        `<code>Fee    :</code> <code>Rp${uniqueNumber}</code>\n` +
                        `<code>Total  :</code> <code>Rp${formatRupiah(uniqueAmount)}</code>\n\n` +
                        `<i>Harap bayar nominal sesuai QRIS.\nPembayaran akan terbatalkan otomatis dalam 5 menit jika tidak ada transaksi masuk.</i>`,
                    parse_mode: 'HTML',
                    reply_markup: {
                        inline_keyboard: [
                            [{ text: '⛔️ Batalkan Transaksi', callback_data: 'cancel_tx' }]
                        ]
                    }
                });

                await createTopup({
                    userId: userData.chatId,
                    trxId: reference,
                    time: Date.now(),
                    amount: amount,
                    method: 'QRIS',
                    status: 'Pending',
                    uniqueAmount: uniqueAmount
                });
                startQrisCheck(chatId, userId, reference, uniqueAmount, 'topup', sentQr.message_id);

                delete userStates[chatId];
                fs.unlinkSync(qrImagePath);

                await updateUser(userData.chatId, { lastMessageId: sentQr.message_id });

            } catch (error) {
                console.error("[QRIS TOPUP ERROR] Gagal membuat/mengirim QRIS:", error.message);

                try { await bot.deleteMessage(chatId, loadingMessage.message_id); } catch (e) { }

                await bot.sendMessage(chatId, `⛔️ Gagal memproses QRIS: ${error.message}`, { parse_mode: 'Markdown' });
                delete userStates[chatId];
            }
        }

        else if (state.step === 'AWAITING_XL_NUMBER') {
            try { await bot.deleteMessage(chatId, userData.lastMessageId); } catch (e) { }
            const normalizedPhone = normalizePhoneNumber(text);

            if (!normalizedPhone || normalizedPhone.length < 10 || normalizedPhone.length > 13) {
                await bot.sendMessage(chatId, "⛔️ Nomor tujuan tidak valid. Mohon masukkan Kembali:", { parse_mode: 'Markdown' });
                return;
            }

            const loadingMessage = await bot.sendMessage(
                chatId,
                `⏳ Sedang melakukan pengecekan kuota untuk nomor \`${normalizedPhone}\`...`,
                { parse_mode: 'Markdown' }
            );

            const messageId = loadingMessage.message_id;

            const result = await cekXL(normalizedPhone);
            let finalMessage;

            if (result.success && result.data) {
                finalMessage =
                    `${separator}\n` +
                    `✅ *Hasil Cek Kuota XL/Axis*\n` +
                    `${separator}\n` +
                    formatResult(result.data) +
                    `${separator}\nCek Kuota By @Gemilangstorebot\n${separator}`;
            } else {
                finalMessage =
                    `${separator}\n❌ *Cek Kuota Gagal*\n${separator}\n` +
                    result.data +
                    `\n${separator}\nCek Kuota By @Gemilangstorebot\n${separator}`;
            }

            await bot.editMessageText(finalMessage, {
                chat_id: chatId,
                message_id: messageId,
                parse_mode: 'Markdown'
            });

            delete userStates[chatId];
            await delay(2000);

            await sendMainMenuFinal(chatId, userData);
        }

        else if (state.step === 'AWAITING_PHONE') {
            try { await bot.deleteMessage(chatId, userData.lastMessageId); } catch (e) { }

            const nameLower = state.productName.toLowerCase();
            let finalDestination = rawText;
            let isGame = nameLower.includes("free fire") || nameLower.includes("mobile legend") || nameLower.includes("mobilelegend");

            if (isGame) {
                finalDestination = rawText.replace(/[^0-9]/g, '');
                if (finalDestination.length < 3) {
                    const sentMsg = await bot.sendMessage(chatId, "⛔️ ID Akun tidak valid. Harap masukkan angka saja.\nMohon Masukkan Kembali:", { parse_mode: 'Markdown' });
                    await updateUser(userData.chatId, { lastMessageId: sentMsg.message_id });
                    return;
                }
            } else {
                const normalized = normalizePhoneNumber(rawText);
                if (!normalized) {
                    const sentMsg = await bot.sendMessage(chatId, "⛔️ Format Nomor Tujuan Tidak Valid (Gunakan 08xx, 628xx, atau +628xx). Mohon Masukkan Kembali:", { parse_mode: 'Markdown' });
                    await updateUser(userData.chatId, { lastMessageId: sentMsg.message_id });
                    return;
                }
                finalDestination = normalized;
            }

            state.phoneNumber = finalDestination;
            state.step = 'AWAITING_CONFIRM';
            userStates[chatId] = state;

            const confirmationMessage = `
${separator}
*Konfirmasi Detail Transaksi*
${separator}
\`Produk :\` \`${escapeHtml(state.productName)}\`
\`Code   :\` \`${escapeHtml(state.productCode)}\`
\`Harga  :\` \`Rp${formatRupiah(state.price)}\`
${separator}
\`Tujuan :\` \`${finalDestination}\`
*Pastikan ${isGame ? 'ID Akun' : 'Nomor Tujuan'} Sudah Benar!*
${separator}`;

            const keyboardOptions = {
                reply_markup: {
                    inline_keyboard: [
                        [
                            { text: '💰 Saldo Bot', callback_data: 'confirm_tx_bot' },
                            { text: '💳 Bayar QRIS', callback_data: 'confirm_tx_qris' }
                        ],
                        [{ text: '⏰ Buat Jadwal', callback_data: 'sched_start' }],
                        [{ text: '⛔️ Batalkan Proses', callback_data: 'cancel_tx' }]
                    ]
                },
                parse_mode: 'Markdown'
            };

            const sentMessage = await bot.sendMessage(chatId, confirmationMessage, keyboardOptions);
            await updateUser(userData.chatId, { lastMessageId: sentMessage.message_id });

        } else if (state.step === 'AWAITING_CONFIRM') {
            await bot.sendMessage(chatId, "⚠️ Mohon tekan tombol *Bayar* atau *Batalkan Proses* di pesan sebelumnya.", { parse_mode: 'Markdown' });
        }
    }
});

const app = express();
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.post('/digiflazz-callback', async (req, res) => {
    const digiSignature = req.headers['x-hub-signature'];
    const secretKey = 'gemilangkinasih';
    const payload = req.rawBody || JSON.stringify(req.body);
    const signature = crypto
        .createHmac('sha1', secretKey)
        .update(payload)
        .digest('hex');
    const selfSignature = 'sha1=' + signature;
    if (digiSignature !== selfSignature) {
        logAction(`[WEBHOOK ATTACK] Signature : ${digiSignature}`);
        logAction(`[WEBHOOK ATTACK] Self Signature : ${selfSignature}`);
        logAction(`[WEBHOOK ATTACK] Signature tidak valid dari IP : ${req.ip}`);
        return res.status(401).send('Invalid Signature');
    }

    await saveCallback(req.body);

    const callbackData = req.body.data;


    res.status(200).send({ status: 'success', message: 'Callback received' });

    if (!callbackData || !callbackData.ref_id) {
        return;
    }

    // [RESELLER MULTI-BOT] Forward webhook ke bot reseller yang sesuai berdasarkan suffix _Sn
    const RESELLER_PORT_MAP = {
        '_S2': 12001,
        '_S3': 12002,
        // Tambahkan reseller baru di sini: '_S4': 12003, '_S5': 12004, dst.
    };

    const matchedSuffix = Object.keys(RESELLER_PORT_MAP).find(suffix => callbackData.ref_id.endsWith(suffix));
    if (matchedSuffix) {
        const targetPort = RESELLER_PORT_MAP[matchedSuffix];
        logAction(`[WEBHOOK FORWARD] Meneruskan webhook ke Bot (suffix: ${matchedSuffix}, port: ${targetPort}) untuk Ref: ${callbackData.ref_id}`);

        // Kirim notif ke grup server.js berdasarkan status transaksi reseller
        if (callbackData.status === 'Sukses' || callbackData.status === 'Gagal') {
            try {
                const [mainTrx] = await pool.query('SELECT * FROM transactions WHERE trxId = ?', [callbackData.ref_id]);
                const resellerUser = mainTrx.length > 0 ? await getUserById(mainTrx[0].userId) : null;
                if (mainTrx.length > 0 && resellerUser) {
                    const date = new Date().toISOString().split('T')[0];
                    const isSukses = callbackData.status === 'Sukses';
                    const statusLabel = isSukses ? 'Status Transaksi Sukses' : 'Status Refund Saldo Sukses';

                    const notifMsg = `
${separator}
${statusLabel}
${separator}
\`✧ Owner :\` \`${escapeHtml(resellerUser.name || resellerUser.username)}\`
\`✧ Code  :\` \`${callbackData.ref_id}\`
\`✧ Info  :\` \`${escapeHtml(mainTrx[0].productName || 'N/A')}\`
\`✧ Via   :\` \`Reseller API\`
\`✧ Pay   :\` \`Rp${formatRupiah(mainTrx[0].price)}\`
\`✧ Date  :\` \`${date}\`
${separator}`.trim();
                    await bot.sendMessage(GROUP_NOTIFICATION_ID, notifMsg, { parse_mode: 'Markdown' }).catch(e => {});
                }
            } catch (e) {
                console.error(`[WEBHOOK RESELLER NOTIF ERROR]: ${e.message}`);
            }
        }

        // Forward ke bot reseller yang sesuai
        axios.post(`http://localhost:${targetPort}/digiflazz-callback`, req.body, {
            headers: { 'x-hub-signature': digiSignature }
        }).catch(e => console.error(`[WEBHOOK FORWARD ERROR] Gagal meneruskan ke port ${targetPort}: ${e.message}`));
        return; // Setop eksekusi di server 1
    }

    const { ref_id, status, message, sn } = callbackData;
    logAction(`WEBHOOK: Callback diterima - Ref: ${ref_id}, Status: ${status}`);

    const foundTransaction = await getTransactionByTrxId(ref_id);
    const foundTopup = foundTransaction ? null : await getTopupByTrxId(ref_id);
    const transactionType = foundTransaction ? 'trx' : (foundTopup ? 'topup' : null);

    if (!transactionType) {
        console.warn(`[WEBHOOK WARNING] Transaksi ${ref_id} tidak ditemukan.`);
        return;
    }

    const currentTrx = foundTransaction || foundTopup;
    const foundUser = await getUserById(currentTrx.userId);
    if (!foundUser) return;

    // Process only if not already in final status
    if (status !== 'Pending' && currentTrx.status !== 'Sukses' && currentTrx.status !== 'Gagal' && !currentTrx.status.includes('Refund')) {
        
        if (status === 'Gagal') {
            if ((currentTrx.method === 'Saldo Bot' || currentTrx.method === 'QRIS' || currentTrx.method === 'Jadwal Otomatis') && transactionType === 'trx') {
                await processRefund(ref_id, foundUser.chatId, currentTrx.price, currentTrx.productName, currentTrx.method, message);
            } else {
                await updateTransaction(ref_id, { status: status, apiMessage: message, sn: sn || 'N/A' });
            }
        } 
        else if (status === 'Sukses' && transactionType === 'topup') {
            const connection = await pool.getConnection();
            try {
                await connection.beginTransaction();
                const [rows] = await connection.query('SELECT status FROM topups WHERE trxId = ? FOR UPDATE', [ref_id]);
                if (rows.length > 0 && rows[0].status === 'Pending') {
                    await updateBalanceAtomic(foundUser.chatId, currentTrx.amount, 'add', connection);
                    await connection.query('UPDATE topups SET status = "Sukses" WHERE trxId = ?', [ref_id]);
                    await connection.commit();
                    
                    const updatedTopup = await getTopupByTrxId(ref_id);
                    await sendTrxNotificationToGroup(updatedTopup, 'topup', foundUser);
                } else {
                    await connection.rollback();
                }
            } catch (e) {
                await connection.rollback();
                console.error("[WEBHOOK ERROR] Topup:", e.message);
            } finally {
                connection.release();
            }
        } 
        else if (status === 'Sukses' && transactionType === 'trx') {
            await updateTransaction(ref_id, { status: status, apiMessage: message, sn: sn || currentTrx.sn });
            const updatedTrx = await getTransactionByTrxId(ref_id);
            await sendTrxNotificationToGroup(updatedTrx, 'trx', foundUser);
        } 
        else {
            if (transactionType === 'trx') {
                await updateTransaction(ref_id, { status: status, apiMessage: message, sn: sn || currentTrx.sn });
            } else {
                await updateTopup(ref_id, { status: status });
            }
        }

        // Final notification for user
        try { await bot.deleteMessage(foundUser.chatId, foundUser.lastMessageId); } catch (e) { }
        const icon = (status === 'Sukses') ? '✅' : '⛔️';
        const updateMessage = `
${separator}
${icon} *Update Cek Status Pesanan*
${separator}
\`Code   :\` \`${ref_id}\`
\`Produk :\` \`${currentTrx.productName || 'N/A'}\`
\`Nomor  :\` \`${currentTrx.phoneNumber || 'N/A'}\`
\`Status :\` \`${status.toUpperCase()}\`
\`Pesan  :\` \`${message}\`
\`Detail :\` \`${sn || 'N/A'}\`
${separator}`;

        await bot.sendMessage(foundUser.chatId, updateMessage, { parse_mode: 'Markdown' });
        await new Promise(resolve => setTimeout(resolve, 2000));

        if (currentTrx.method !== 'Jadwal Otomatis') {
            const updatedUser = await getUserById(foundUser.chatId);
            await sendMainMenuFinal(foundUser.chatId, updatedUser);
        }
    }
});

    logAction(`WEBHOOK: Express server listening on port ${WEBHOOK_PORT}`);

function startJadwalWorker() {
    schedule.scheduleJob('* * * * *', async function () {
        const nowWIB = new Date(new Date().toLocaleString("en-US", { timeZone: "Asia/Jakarta" }));
        const curD = nowWIB.getFullYear() + '-' +
            String(nowWIB.getMonth() + 1).padStart(2, '0') + '-' +
            String(nowWIB.getDate()).padStart(2, '0');
        const curT = String(nowWIB.getHours()).padStart(2, '0') + ':' +
            String(nowWIB.getMinutes()).padStart(2, '0');

        // Check for due schedules
        const dueSchedules = await getDueSchedules(curD, curT);

        for (const j of dueSchedules) {
            const user = await getUserById(j.userId);
            if (!user) continue;

            const productResult = findProductByName(j.productName);
            const latestProduct = productResult ? productResult.data : null;

            if (!latestProduct) {
                const failCode = `FAIL${Date.now()}`;

                await createTransaction({
                    userId: j.userId,
                    trxId: failCode,
                    time: Date.now(),
                    productName: j.productName,
                    phoneNumber: j.phoneNumber,
                    price: j.price,
                    method: 'Jadwal Otomatis',
                    status: 'Gagal (Produk Tidak Tersedia)',
                    sn: 'Produk sudah tidak tersedia'
                });

                await bot.sendMessage(
                    j.userId,
                    `
${separator}
⛔️ *Status Jadwal Otomatis*
${separator}
\`Code   :\` \`${failCode}\`
\`Produk :\` \`${j.productName}\`
\`Nomor  :\` \`${j.phoneNumber}\`
\`Waktu  :\` \`${formatTglJadwal(j.executeDate, j.executeTime)}\`
\`Status :\` \`GAGAL\`
\`Harga  :\` \`Rp${formatRupiah(j.price)}\`
\`Saldo  :\` \`Rp${formatRupiah(user.saldo)}\`
\`Detail :\` \`Produk sudah tidak tersedia\`
${separator}`,
                    { parse_mode: 'Markdown' }
                );

                await deleteSchedule(j.id);
                continue;
            }

            const currentPrice = latestProduct.sellingPrice;

            if (parseFloat(user.saldo) >= currentPrice) {
                try {
                    const refId = `SCH${Date.now()}`;
                    
                    // Deduct balance first
                    await updateBalanceAtomic(j.userId, currentPrice, 'subtract');

                    const res = await callDigiflazzAPI(refId, latestProduct.code, j.phoneNumber);

                    if (res.status === 'Gagal') {
                        // Refund immediately if API returns Gagal
                        await processRefund(refId, j.userId, currentPrice, j.productName, 'Jadwal Otomatis', res.message);
                    } else {
                        await createTransaction({
                            userId: j.userId,
                            trxId: refId,
                            time: Date.now(),
                            productName: j.productName,
                            productCode: j.productCode,
                            phoneNumber: j.phoneNumber,
                            price: currentPrice,
                            method: 'Jadwal Otomatis',
                            status: res.status,
                            sn: res.sn || res.message
                        });

                        const updatedUser = await getUserById(j.userId);
                        const successMsg = `
${separator}
✅ *Status Jadwal Otomatis*
${separator}
\`Code   :\` \`${refId}\`
\`Produk :\` \`${j.productName}\`
\`Nomor  :\` \`${j.phoneNumber}\`
\`Waktu  :\` \`${formatTglJadwal(j.executeDate, j.executeTime)}\`
\`Status :\` \`${res.status.toUpperCase()}\`
\`Harga  :\` \`Rp${formatRupiah(currentPrice)}\`
\`Saldo  :\` \`Rp${formatRupiah(updatedUser.saldo)}\`
${separator}`;

                        await bot.sendMessage(j.userId, successMsg, { parse_mode: 'Markdown' }).catch(() => {});
                    }

                } catch (err) {
                    const failCode = `FAIL${Date.now()}`;
                    await createTransaction({
                        userId: j.userId,
                        trxId: failCode,
                        time: Date.now(),
                        productName: j.productName,
                        productCode: j.productCode,
                        phoneNumber: j.phoneNumber,
                        price: currentPrice,
                        method: 'Jadwal Otomatis',
                        status: 'Gagal (Error API)',
                        sn: err.message
                    });

                    const failApiMsg = `
${separator}
⛔️ *Status Jadwal Otomatis*
${separator}
\`Code   :\` \`${failCode}\`
\`Produk :\` \`${j.productName}\`
\`Nomor  :\` \`${j.phoneNumber}\`
\`Waktu  :\` \`${formatTglJadwal(j.executeDate, j.executeTime)}\`
\`Status :\` \`GAGAL\`
\`Harga  :\` \`Rp${formatRupiah(currentPrice)}\`
\`Saldo  :\` \`Rp${formatRupiah(user.saldo)}\`
\`Detail :\` \`Error API (${err.message})\`
${separator}`;

                    await bot.sendMessage(j.userId, failApiMsg, { parse_mode: 'Markdown' });
                }
            } else {
                const failCode = `FAIL${Date.now()}`;
                await createTransaction({
                    userId: j.userId,
                    trxId: failCode,
                    time: Date.now(),
                    productName: j.productName,
                    phoneNumber: j.phoneNumber,
                    price: currentPrice,
                    method: 'Jadwal Otomatis',
                    status: 'Gagal (Saldo Kurang)',
                    sn: 'Saldo tidak mencukupi'
                });

                const failSaldoMsg = `
${separator}
⛔️ *Status Jadwal Otomatis*
${separator}
\`Code   :\` \`${failCode}\`
\`Produk :\` \`${j.productName}\`
\`Nomor  :\` \`${j.phoneNumber}\`
\`Waktu  :\` \`${formatTglJadwal(j.executeDate, j.executeTime)}\`
\`Status :\` \`GAGAL\`
\`Harga  :\` \`Rp${formatRupiah(currentPrice)}\`
\`Saldo  :\` \`Rp${formatRupiah(user.saldo)}\`
\`Detail :\` \`Saldo tidak mencukupi\`
${separator}`;

                await bot.sendMessage(j.userId, failSaldoMsg, { parse_mode: 'Markdown' });
            }
            await deleteSchedule(j.id);
        }

        // Cleanup old schedules
        const currentDateTime = nowWIB.toISOString().slice(0, 16).replace('T', ' ');
        const oldSchedules = await getOldSchedules(currentDateTime);

        for (const oldSched of oldSchedules) {
            logAction(`[CLEANUP] Menghapus jadwal basi user ${oldSched.userId} : ${oldSched.productName} (${oldSched.executeDate} ${oldSched.executeTime})`);
            await deleteSchedule(oldSched.id);
        }
    });
}
